cmake_minimum_required(VERSION 3.16)
project(ObligationDynamics VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GEOMETRY ${CMAKE_CURRENT_SOURCE_DIR}/../wrench_transport/phys)
add_library(obligation_geometry STATIC ${GEOMETRY}/hull.cpp ${GEOMETRY}/collide.cpp)
target_include_directories(obligation_geometry PUBLIC ${GEOMETRY})
add_library(obligation_dynamics STATIC engine.cpp)
target_link_libraries(obligation_dynamics PUBLIC obligation_geometry)
target_include_directories(obligation_dynamics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(obligation_dynamics PRIVATE /W4 /fp:precise)
else()
  target_compile_options(obligation_geometry PUBLIC -ffp-contract=off)
  target_compile_options(obligation_dynamics PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(obligation_tests tests.cpp)
target_link_libraries(obligation_tests PRIVATE obligation_dynamics)
add_test(NAME obligation_tests COMMAND obligation_tests)
add_executable(obligation_minimal minimal.cpp)
target_link_libraries(obligation_minimal PRIVATE obligation_dynamics)
add_test(NAME obligation_minimal COMMAND obligation_minimal)
add_executable(obligation_bench bench.cpp)
target_link_libraries(obligation_bench PRIVATE obligation_dynamics)
set(WRENCH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WRENCH_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory(${GEOMETRY} legacy EXCLUDE_FROM_ALL)
add_executable(obligation_activity activity_bench.cpp)
target_link_libraries(obligation_activity PRIVATE obligation_dynamics Wrench::Physics)
