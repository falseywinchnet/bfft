cmake_minimum_required(VERSION 3.16)
project(TapeDynamics VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
find_package(Threads REQUIRED)
add_library(atomic_pool STATIC threadpool/src/threadpool_atomic_fast.c)
target_include_directories(atomic_pool PUBLIC threadpool/include)
target_link_libraries(atomic_pool PUBLIC Threads::Threads)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GEOMETRY ${CMAKE_CURRENT_SOURCE_DIR}/../wrench_transport/phys)
add_library(tape_geometry STATIC ${GEOMETRY}/hull.cpp ${GEOMETRY}/collide.cpp)
target_include_directories(tape_geometry PUBLIC ${GEOMETRY})
add_library(tape_dynamics STATIC engine.cpp)
target_link_libraries(tape_dynamics PUBLIC tape_geometry atomic_pool)
target_include_directories(tape_dynamics PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(tape_dynamics PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(tape_geometry PUBLIC -ffp-contract=off)
enable_testing()
add_executable(tape_tests tests.cpp)
target_link_libraries(tape_tests PRIVATE tape_dynamics)
add_test(NAME tape_tests COMMAND tape_tests)
add_executable(tape_bench bench.cpp)
target_link_libraries(tape_bench PRIVATE tape_dynamics)

option(TAPE_BUILD_BASELINE "Build the unchanged Wrench comparison" ON)
if(TAPE_BUILD_BASELINE)
  set(WRENCH_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(WRENCH_BUILD_TESTS ON CACHE BOOL "" FORCE)
  add_subdirectory(${GEOMETRY} wrench_baseline)
  add_executable(tape_legacy_bench legacy_bench.cpp)
  target_link_libraries(tape_legacy_bench PRIVATE Wrench::Physics)
endif()
